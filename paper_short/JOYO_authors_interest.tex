\subsection{Authors Affiliations and Research
Interests}\label{authors-affiliations-and-research-interests}

\textbf{Andreas Laffert}, Institute of Sociology, Pontifical Catholic
University of Chile, Santiago de Chile. Research interests: social
stratification, social mobility, wealth, and attitudes towards the
marketization of social welfare.

\textbf{Juan Carlos Castillo}, Departament of Sociology, University of
Chile, Santiago de Chile. Research interests: distributive justice from
a quantitative perspective, using multilevel methods, structural
equation modeling, and open science tools.

\textbf{René Canales}, Institute of Sociology, Pontifical Catholic
University of Chile, Santiago de Chile. Research interests: contentious
politics, political attitudes, and social stratification.

\textbf{Tomás Urzúa}, Departament of Sociology, University of Chile,
Santiago de Chile. Research interests: sociology of education,
meritocracy, and quantitative methods.

\textbf{Kevin Carrasco}, Departament of Sociology, University of Chile,
Santiago de Chile. Research interests: inequality, distributive justice,
civic education, open science, quantitative methods in social research,
and survey design.
